Code to Load the DefaultPawnClass (BP) from C++ Game Mode:

void ATAB_EEGameMode::InitGame(const FString& MapName, const FString& Options, FString& ErrorMessage)
{
    Super::InitGame(MapName, Options, ErrorMessage);

    // Load the asset safely during InitGame instead of the constructor
    FSoftClassPath PawnClassPath(TEXT("/Game/Blueprints/SandboxCharacter_Mover_Ragdoll.SandboxCharacter_Mover_Ragdoll_C"));
    UClass* LoadedClass = PawnClassPath.TryLoadClass<APawn>();

    if (LoadedClass)
    {
        DefaultPawnClass = LoadedClass;
    }
}
